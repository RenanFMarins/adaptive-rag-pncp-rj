def _relevancia_textual(pergunta: str, conteudo: str) -> float:

    """SobreposiÃ§Ã£o simples de termos informativos; nÃ£o substitui busca semÃ¢ntica."""

    ignorar = {

        "quais", "qual", "que", "como", "sao", "a", "o", "as", "os",

        "de", "do", "da", "dos", "das", "em", "na", "no", "para",

        "previstas", "previstos", "prevista", "previsto", "lei", "decreto",

    }

    termos = set(re.findall(r"[a-z]+", normalizar_para_busca(pergunta))) - ignorar

    if not termos:

        return 0.0

    palavras = set(re.findall(r"[a-z]+", normalizar_para_busca(conteudo)))

    return len(termos & palavras) / len(termos)





def _reordenar_normativa(

    pergunta: str,

    candidatos: list[tuple[str, float]],

    chunks: dict[str, DocumentChunk],

    documento_id: int,

) -> list[tuple[str, float]]:

    """Prioriza a fonte solicitada; mantÃ©m o score RRF intacto."""

    return sorted(

        candidatos,

        key=lambda item: (

            int(chunks[item[0]].document_id == documento_id),

            _relevancia_textual(pergunta, chunks[item[0]].content),

            item[1],

            item[0],

        ),

        reverse=True,

    )





def _buscar_no_documento(

    db: Session,

    pergunta: str,

    documento_id: int,

    allowed_chunk_ids: set[str] | None,

    limite: int,

) -> list[tuple[str, float]]:

    """Recupera evidÃªncias do documento citado sem alterar Ã­ndices ou dados.



    A pontuaÃ§Ã£o local serve para ordenar candidatos; nÃ£o Ã© um score BM25.

    """

    if limite <= 0:

        return []

    consulta = (

        select(DocumentChunk)

        .where(DocumentChunk.document_id == documento_id)

        .order_by(DocumentChunk.id)

    )

    if allowed_chunk_ids is not None:

        consulta = consulta.where(DocumentChunk.chunk_id.in_(allowed_chunk_ids))

    candidatos = list(db.scalars(consulta))

    if not candidatos:

        return []



    def normalizar_lexico(texto: str) -> str:
        # NFKD converte inclusive "Ã§" em "c"; evita "atribui" em vez
        # de "atribuicoes" quando extraÃ­mos palavras com [a-z].
        decomposed = unicodedata.normalize("NFKD", texto or "")
        return "".join(c for c in decomposed if not unicodedata.combining(c)).lower()

    pergunta_sem_norma = re.sub(
        r"\b(?:lei|decreto)(?:\s+(?:estadual|federal|municipal|rj))*"
        r"\s+(?:n[ÂºÂ°o.]*)?\s*\d{1,3}(?:\.\d{3})*(?:/\d{4})?",
        " ", normalizar_lexico(pergunta),
    )
    ignorar = {
        "quais", "qual", "como", "que", "dos", "das", "nos", "nas",
        "sao", "previstas", "previstos", "sobre", "para", "pela", "pelo",
        "contrato", "contratos", "decreto", "estadual", "rj", "numero",
    }
    equivalencias = {
        "fiscais": "fiscal", "fiscalizacao": "fiscal",
        "atribuicoes": "competencia", "atribuicao": "competencia",
        "competencias": "competencia", "compete": "competencia",
        "responsabilidades": "competencia", "responsabilidade": "competencia",
        "deveres": "competencia", "dever": "competencia",
        "obrigacoes": "obrigacao", "obrigacao": "obrigacao",
    }

    def tokens(texto: str) -> set[str]:
        return {
            equivalencias.get(t, t)
            for t in re.findall(r"[a-z]{3,}", normalizar_lexico(texto))
        }

    termos = {
        equivalencias.get(t, t)
        for t in re.findall(r"[a-z]{3,}", pergunta_sem_norma)
        if t not in ignorar
    }

    def pontuar(chunk: DocumentChunk) -> float:
        conteudo = chunk.content or ""
        palavras = tokens(conteudo)
        inicio = normalizar_lexico(conteudo)
        inicio = re.sub(
            r"^\s*(?:\[)?art\.?\s*\d+\s*(?:[ÂºÂ°])?"
            r"\s*(?:\(continuacao\)\])?\s*", "", inicio,
        )
        # O caput costuma indicar a matÃ©ria regulada pelo artigo.
        cabecalho = re.split(r"[.;:\n]", inicio, maxsplit=1)[0][:240]
        termos_cabecalho = tokens(cabecalho)
        encontrados = termos & palavras
        score = sum(2.0 if len(t) >= 6 else 1.0 for t in encontrados)
        score += 5.0 * len(termos & termos_cabecalho)
        if "competencia" in termos and "competencia" in termos_cabecalho:
            score += 8.0
        if "preambulo" in chunk.chunk_id.lower():
            score *= 0.25
        return score

    ordenados = sorted(

        candidatos,

        key=lambda chunk: (pontuar(chunk), -chunk.id),

        reverse=True,

    )

    # NÃ£o introduzir chunks sem correspondÃªncia lexical, salvo quando a

    # pergunta nÃ£o possui termos informativos apÃ³s a normalizaÃ§Ã£o.

    escolhidos = [c for c in ordenados if pontuar(c) > 0] if termos else ordenados

    return [(c.chunk_id, pontuar(c)) for c in escolhidos[:limite]]
